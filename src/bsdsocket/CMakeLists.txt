# bsdsocket.library, the AmiTCP/Roadshow compatibility surface.
#
# Built as a static library here; the final shared-library link happens once
# the ThreadX Exec port, the SANA-II shim and the netstack singleton exist
# (docs/RESEARCH.md S8, milestones 1-3). See AMINETXDUO_BSDSOCKET_LINK below.

option(AMINETXDUO_BSDSOCKET_LINK
       "Link LIBS:bsdsocket.library as an AmigaOS shared library" ON)

set(BSDSOCKET_SOURCES
    library.c               # romtag, Open/Close/Expunge, child bases
    bsdsocket_vectors.c     # GENERATED by tools/gen_vectors.py
    socket.c
    raw.c                   # SOCK_RAW over the NetX Duo IP raw packet filter
    oob.c                   # MSG_OOB / SIOCATMARK: TCP urgent data
    transfer.c
    rxdirect.c
    udp_queue.c             # queued UDP header layout, host-regression tested
    cmsg.c                  # RFC 3542 ancillary data: PKTINFO, HOPLIMIT,
                            # ICMP6_FILTER, and the IPv4 half of the same
    options.c
    select.c
    errno.c
    inet.c
    resolver.c
    netdb.c                 # get{serv,proto,net}by* over src/config/netdb.c
    addrinfo.c              # getaddrinfo/getnameinfo/freeaddrinfo/gai_strerror
    handoff.c               # ObtainSocket/ReleaseSocket cross-base transfer
    bpf.c                   # the eight bpf_* LVOs over src/bpf/
    netstatus.c             # NetStackQuery/NetStackControl: the live stack,
                            # readable and steerable from a Shell command
    roadshow.c              # the Tier 3 extensions whose ABI is pinned down
    interfaces.c            # ObtainInterfaceList/QueryInterfaceTagList/
                            # ConfigureInterfaceTagList, from the NDK 3.2 autodoc
    addralloc.c             # Create/DeleteAddrAllocMessage: the message
                            # BeginInterfaceConfig takes
    netmonitor.c            # AddNetMonitorHookTagList/RemoveNetMonitorHook
    routing.c               # AddRoute/DeleteRoute/GetRouteInfo, same source
    netstats.c              # GetNetworkStatistics: 4.4BSD's ipstat/tcpstat/
                            # udpstat/icmpstat over NetX Duo's counters
    netx_call.c             # ThreadX context bracket for NetX Duo calls
    library_runtime.c)      # DOSBase, rand(): no crt0 in a shared library

# in6.c is the AF_INET6 ABI: the _Static_asserts that pin sockaddr_in6, the
# address conversions and the IPPROTO_IPV6 options. It has no IPv4 content at
# all, so the floor build simply does not compile it.
if(AMINETXDUO_IPV6)
    list(APPEND BSDSOCKET_SOURCES in6.c)
endif()


# tcp_handler.c is the TCP: DOS handler: a socket reachable as a file handle,
# so `Type TCP:host/port` works and a program that speaks DOS files can speak
# the network. Nothing in the library depends on it, library.c starts it and
# asks whether it is alive, and both are compiled out with it.
if(AMINETXDUO_TCPDEVICE)
    list(APPEND BSDSOCKET_SOURCES tcp_handler.c)
endif()

# nxcontext.c is the ONE private LVO tls.library needs (see
# include/aminetxduo/nxcontext.h). It exists only in an AMINETXDUO_TLS build,
# both as an object and as a vector-table slot, so that a default build of
# bsdsocket.library is byte-identical to a build of a tree without any of this.
if(AMINETXDUO_TLS)
    list(APPEND BSDSOCKET_SOURCES nxcontext.c)
endif()

# mcast.c is the RFC 1112 membership surface. It is the only thing that calls
# NetX Duo's IGMP services, so leaving it out is what keeps the nx_igmp_*
# objects out of the link, see the floor tier in dist/make-dist.sh.
if(AMINETXDUO_MULTICAST)
    list(APPEND BSDSOCKET_SOURCES mcast.c)
endif()

add_library(aminetxduo_bsdsocket STATIC ${BSDSOCKET_SOURCES})

target_include_directories(aminetxduo_bsdsocket PUBLIC
    ${AMINETXDUO_CORE_INCLUDES}
    "${CMAKE_CURRENT_SOURCE_DIR}")

target_link_libraries(aminetxduo_bsdsocket PUBLIC
    aminetxduo_common netxduo aminetxduo_netstack aminetxduo_config)

if(AMINETXDUO_BPF)
    target_link_libraries(aminetxduo_bsdsocket PUBLIC aminetxduo_bpf)
endif()

# A shared library has no startup code and must not drag in newlib's stdio.
target_compile_options(aminetxduo_bsdsocket PRIVATE -Wall -Wextra)

if(AMINETXDUO_TLS)
    target_compile_definitions(aminetxduo_bsdsocket PUBLIC AMINETXDUO_TLS_CONTEXT=1)
endif()

# ---------------------------------------------------------------- generator --
#
# bsdsocket_vectors.[ch] are checked in so a normal build needs no Python.
# Re-run after touching the IMPLEMENTED set in tools/gen_vectors.py.

find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
    add_custom_target(bsdsocket-vectors
        COMMAND ${Python3_EXECUTABLE}
                "${CMAKE_SOURCE_DIR}/tools/gen_vectors.py"
                --ndk "${AMIGA_NDK_INCLUDE}"
                --outdir "${CMAKE_CURRENT_SOURCE_DIR}"
        COMMENT "Regenerating the bsdsocket.library LVO vector table"
        VERBATIM)
endif()

# --------------------------------------------------------------------- link --
#
# library.c MUST come first so its "moveq #-1,d0 / rts" lands at offset 0 of
# the first code hunk, that is what makes the library file safe to execute.
# -nostartfiles because a shared library has no crt0. Note that -noixemul is
# NOT usable with this newlib toolchain (docs/RESEARCH.md S5.4).

if(AMINETXDUO_BSDSOCKET_LINK)
    # The sources are listed directly rather than linked from the archive: an
    # archive member is only pulled in when something references it, and NOTHING
    # references the romtag, it is found by Exec scanning the loaded segment.
    add_executable(bsdsocket_library ${BSDSOCKET_SOURCES})
    set_target_properties(bsdsocket_library PROPERTIES
        OUTPUT_NAME "bsdsocket.library")

    # Listing the sources directly, above, is what stops the ARCHIVE dropping
    # the romtag.  Under -flto that is not enough: the whole-program view sees
    # nothing referencing bsd_romtag either and removes it, the autoinit table,
    # the vector table and every vector, and the link still succeeds.  Naming
    # it holds the whole chain.  See the -flto note in the top-level
    # CMakeLists.txt.
    if(CMAKE_CROSSCOMPILING)
        target_link_options(bsdsocket_library PRIVATE "-Wl,-u,_bsd_romtag")
    endif()

    target_include_directories(bsdsocket_library PRIVATE
        ${AMINETXDUO_CORE_INCLUDES} "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(bsdsocket_library PRIVATE
        aminetxduo_netstack aminetxduo_sana2 aminetxduo_config
        aminetxduo_common netxduo_addons netxduo threadx threadx_port)

    if(AMINETXDUO_BPF)
        target_link_libraries(bsdsocket_library PRIVATE aminetxduo_bpf)
    endif()

    if(AMINETXDUO_TLS)
        target_compile_definitions(bsdsocket_library PRIVATE
                                   AMINETXDUO_TLS_CONTEXT=1)
    endif()

    # -nostartfiles, not -nostdlib: there is no crt0 in a shared library, but
    # ThreadX and NetX Duo do want memset/memcpy/memcmp and gcc wants its
    # integer-divide helpers, and the linker only pulls the archive members that
    # are actually referenced, no stdio comes along for the ride.
    #
    # The map is for tools/profiler/profreport.py --lib.  A profile of a real
    # application samples this library as a shared library, and the linker
    # leaves file statics out of HUNK_SYMBOL entirely, their samples land
    # silently on the preceding global.  The map's placement of each object,
    # plus nm on that object, is the only route to them, and it is roughly half
    # the symbols here.  It does not change the binary.
    # --gc-sections, which src/tools/ has had all along and the three shared
    # libraries never did.  9,804 bytes off bsdsocket.library, and it is what
    # makes the -ffunction-sections src/config/ already pays for do something
    # here rather than only in the commands: that comment said "the library
    # link does not collect, so all it pays is per-section padding, 1,848
    # bytes", and this is the other half of that sentence arriving.
    target_link_options(bsdsocket_library PRIVATE -nostartfiles
        -Wl,--gc-sections
        "-Wl,-Map=$<TARGET_FILE_DIR:bsdsocket_library>/bsdsocket.library.map")

    # 39,948 bytes, 11% of the library, and nothing on the Amiga reads them:
    # profreport.py --lib works from the map above and the objects, not from
    # HUNK_SYMBOL, which is why the map is written before the strip runs.
    if(CMAKE_STRIP AND NOT AMINETXDUO_KEEP_SYMBOLS)
        add_custom_command(TARGET bsdsocket_library POST_BUILD
            COMMAND "${CMAKE_STRIP}" "$<TARGET_FILE:bsdsocket_library>"
            VERBATIM)
    endif()

    # The "moveq #-1,d0 / rts" at the top of library.c has to land at offset 0
    # of the first code hunk, and the only thing putting it there is this
    # object's position on the link line -- the order of a source list nobody
    # is looking at.  See cmake/check-library-entry.cmake.
    add_custom_command(TARGET bsdsocket_library POST_BUILD
        COMMAND "${CMAKE_COMMAND}"
                "-DLIBRARY_BINARY=$<TARGET_FILE:bsdsocket_library>"
                -P "${CMAKE_SOURCE_DIR}/cmake/check-library-entry.cmake"
        VERBATIM)

    # The $VER: string, checked the way every command in src/tools is.  It is
    # found by AmigaDOS scanning the file, so no reference to it exists that
    # the linker can see, and this target links --gc-sections.  `used' on
    # bsd_lib_ver keeps it in the IR and says nothing about whether it
    # survives to the file.  Nothing else would notice its absence: `version
    # full file LIBS:bsdsocket.library' would simply report nothing.
    add_custom_command(TARGET bsdsocket_library POST_BUILD
        COMMAND "${CMAKE_COMMAND}"
                "-DTOOL_BINARY=$<TARGET_FILE:bsdsocket_library>"
                -P "${CMAKE_SOURCE_DIR}/cmake/check-version-tag.cmake"
        VERBATIM)
endif()
